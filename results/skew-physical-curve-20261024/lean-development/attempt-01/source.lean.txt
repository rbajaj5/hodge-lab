import HodgeLab.OneReflectionSkew.PhysicalSourceNormalization
import Mathlib.FieldTheory.IsAlgClosed.AlgebraicClosure
import Mathlib.RingTheory.AlgebraicIndependent.Basic
import Mathlib.RingTheory.Localization.Away.Basic

/-! An actual algebraic realization of the physical chart with independent base coordinates.
The specialization domain is the polynomial ring, then its z-localization;
there is deliberately no specialization of the whole ambient fraction field. -/
namespace HodgeLab.OneReflectionSkew.PhysicalCurveRealization
open MvPolynomial ChartSubstitution NormalChart ResidueNormalization
open PhysicalSourceNormalization
noncomputable section

abbrev BasePoly := MvPolynomial (Fin 4) ℚ
abbrev Base := FractionRing BasePoly
instance baseCharZero : CharZero Base :=
  charZero_of_injective_algebraMap (IsFractionRing.injective BasePoly Base)
abbrev Curve := AlgebraicClosure Base

/-- The four coordinates remain algebraically independent in the realization. -/
def param (i : Fin 4) : Curve := algebraMap BasePoly Curve (X i)
abbrev U : Curve := param 0
abbrev r : Curve := param 1
abbrev w : Curve := param 2
abbrev t : Curve := param 3

theorem base_embedding_injective : Function.Injective (algebraMap Base Curve) :=
  (algebraMap Base Curve).injective

theorem polynomial_embedding_injective : Function.Injective (algebraMap BasePoly Curve) := by
  rw [IsScalarTower.algebraMap_eq BasePoly Base Curve]
  exact base_embedding_injective.comp (IsFractionRing.injective BasePoly Base)

theorem parameters_independent : AlgebraicIndependent ℚ param := by
  exact (MvPolynomial.algebraicIndependent_X (Fin 4) ℚ).map'
    (f := IsScalarTower.toAlgHom ℚ BasePoly Curve) polynomial_embedding_injective

theorem param_ne_zero (i : Fin 4) : param i ≠ 0 := parameters_independent.ne_zero i

/-- Seven times the literal cubic, represented before passage to any field. -/
def cubic7 : BasePoly :=
  7*(1+(X 1)^6+(X 2)^6)*(X 0)^3 +
  7*X 1*X 3*(1+(X 1)^2*X 2)*(X 0)^2 +
  X 1*X 3*(1-(X 3)^2)*X 0+7*(1+(X 3)^6)

theorem cubic7_ne_zero : cubic7 ≠ 0 := by
  intro h
  have hh := congrArg (MvPolynomial.eval (fun _ : Fin 4 => (0 : ℚ))) h
  norm_num [cubic7] at hh

theorem cubic7_image : algebraMap BasePoly Curve cubic7 = 7*cubic U r w t := by
  simp only [cubic7, map_add, map_mul, map_pow, map_sub, map_one, map_ofNat]
  simp only [cubic, aCoeff, bCoeff, cCoeff, dCoeff, U, r, w, t, param]
  ring

theorem cubic_ne_zero : cubic U r w t ≠ 0 := by
  intro h
  have he : algebraMap BasePoly Curve cubic7 = 0 := by rw [cubic7_image,h,mul_zero]
  exact cubic7_ne_zero (polynomial_embedding_injective (he.trans (map_zero _).symm))

def z : Curve := (IsAlgClosed.exists_pow_nat_eq (-cubic U r w t)
  (by norm_num : 0 < (6 : ℕ))).choose

theorem z_power : z^6 = -cubic U r w t :=
  (IsAlgClosed.exists_pow_nat_eq (-cubic U r w t)
    (by norm_num : 0 < (6 : ℕ))).choose_spec

theorem curve_equation : z^6+cubic U r w t = 0 := by rw [z_power]; ring

theorem z_ne_zero : z ≠ 0 := by
  intro hz
  have h := curve_equation
  rw [hz] at h
  simpa using cubic_ne_zero h

/-- Recover the original square-cover coordinate; no new parameter is collapsed. -/
def x : Curve := (IsAlgClosed.exists_pow_nat_eq U (by norm_num : 0 < (2 : ℕ))).choose

theorem x_square : x^2 = U :=
  (IsAlgClosed.exists_pow_nat_eq U (by norm_num : 0 < (2 : ℕ))).choose_spec

theorem x_ne_zero : x ≠ 0 := by
  intro hx
  have h := x_square
  rw [hx, zero_pow (by decide : (2:ℕ) ≠ 0)] at h
  exact param_ne_zero 0 h.symm

theorem protected_F_zero : CoordinateComparison.skewF (physicalPoint x r w t z) = 0 := by
  rw [protected_substitution, x_square]
  exact curve_equation

theorem protected_A_image : CoordinateComparison.skewA (physicalPoint x r w t z) = x^3 :=
  protected_source_substitution x r w t z

abbrev P := AmbientDerivation.P

def coords : Fin 5 → Curve := ![U,r,w,t,z]
def chartMap : P →ₐ[ℚ] Curve := MvPolynomial.aeval coords

@[simp] theorem chartMap_X (i : Fin 5) : chartMap (X i) = coords i := by
  simp [chartMap]

theorem chartMap_relation : chartMap (chartPolynomial (K:=ℚ)) = 0 := by
  have hc : chartMap (C (1/7 : ℚ)) = (1/7 : Curve) := by simp [chartMap]
  simp only [chartPolynomial,map_add,map_mul,map_pow,map_sub,map_one,hc,chartMap_X]
  change z^6+(1+r^6+w^6)*U^3+r*t*(1+r^2*w)*U^2+(1/7:Curve)*r*t*(1-t^2)*U+(1+t^6)=0
  convert curve_equation using 1 <;> simp only [cubic,aCoeff,bCoeff,cCoeff,dCoeff] <;> ring

/-- An ambient fraction-field specialization killing G would contradict injectivity. -/
theorem no_ambient_field_specialization (f : AmbientDerivation.K →+* Curve)
    (h : f AmbientDerivation.G = 0) : False :=
  AmbientDerivation.G_ne_zero (f.injective (h.trans (map_zero f).symm))

end
end HodgeLab.OneReflectionSkew.PhysicalCurveRealization
