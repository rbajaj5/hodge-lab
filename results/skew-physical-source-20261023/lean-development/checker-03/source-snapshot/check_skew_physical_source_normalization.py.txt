def write(name, value):
    (OUT/name).write_text(json.dumps(value, ensure_ascii=False, indent=2)+'\n',
                          encoding='utf-8', newline='\n')


def raw_digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def run(label, command, reject=False):
    remaining = 600 - (time.monotonic()-START)
    require(remaining > 1, '600-second budget exhausted')
    bound = min(120, int(remaining))
    execution = command
    if os.name == 'nt':
        wroot = '/mnt/'+ROOT.drive[0].lower()+ROOT.as_posix()[2:]
        execution = ['wsl.exe', '--cd', wroot, '-e', 'sh', '-lc',
                     'exec timeout '+str(bound)+' '+shlex.join(command)]
    began = time.monotonic()
    started = datetime.datetime.now(datetime.timezone.utc).isoformat()
    try:
        completed = subprocess.run(execution, cwd=ROOT, capture_output=True,
                                   encoding='utf-8', timeout=bound)
        code, stdout, stderr = completed.returncode, completed.stdout, completed.stderr
    except subprocess.TimeoutExpired as error:
        def text(value):
            return value.decode('utf-8', errors='replace') if isinstance(value, bytes) else value or ''
        code, stdout, stderr = 124, text(error.stdout), text(error.stderr)
    log = stdout+stderr
    for suffix, value in [('stdout.log', stdout), ('stderr.log', stderr), ('log', log)]:
        (OUT/(label+'.'+suffix)).write_text(value, encoding='utf-8', newline='\n')
    RECORDS.append(dict(label=label, command=command, execution_command=execution,
        cwd='formal', started_utc=started, bound_seconds=bound, exit_code=code,
        elapsed_seconds=time.monotonic()-began, expected_failure=reject,
        log=label+'.log', log_lf_sha256=digest(OUT/(label+'.log')),
        stdout=label+'.stdout.log', stdout_sha256=raw_digest(OUT/(label+'.stdout.log')),
        stderr=label+'.stderr.log', stderr_sha256=raw_digest(OUT/(label+'.stderr.log'))))
    write('commands.json', RECORDS)
    print(json.dumps({'label':label, 'exit_code':code,
                      'seconds':RECORDS[-1]['elapsed_seconds']}), flush=True)
    require(code != 124, label+' timed out; interrupted calculations are not results')
    if reject:
        require(code != 0 and log.count('error:') == 1 and 'unsolved goals' in log and '⊢ False' in log,
                label+' did not fail at the intended false mathematical goal: '+log[:2500])
        require('failed to synthesize' not in log and 'unknown' not in log.lower(),
                'Unrelated control error: '+label)
    else:
        require(code == 0, label+' failed: '+log[:2500])
    return log


def checked_inputs():
    inputs = json.loads((HISTORY/'inputs.json').read_text(encoding='utf-8'))
    ref = inputs['source_pins_manifest']
    source_manifest = (ROOT.parent/ref['path']).resolve()
    require(source_manifest.is_relative_to(ROOT.parent.resolve()), 'Source manifest path escapes repository')
    require(raw_digest(source_manifest) == ref['sha256'], 'Source-pin manifest changed')
    pins = json.loads(source_manifest.read_text(encoding='utf-8'))
    require(pins['parent'] == inputs['notebook_parent'], 'Notebook parent mismatch')
    require(pins['upstream'] == inputs['upstream_commit'], 'Upstream pin mismatch')
    require(len(pins['files']) == 44, 'Missing pinned source/receipt inputs')
    for name, pin in pins['files'].items():
        path = (ROOT.parent/name).resolve()
        require(path.is_relative_to(ROOT.parent.resolve()), 'Pinned path escapes repository')
        require(raw_digest(path) == pin['sha256'], 'Pinned source or historical receipt changed: '+name)
        require(path.stat().st_size == pin['bytes'], 'Pinned byte count changed: '+name)
    prior = json.loads((ROOT.parent/inputs['reused_scalar_extension_receipt']).read_text(encoding='utf-8'))
    require(prior['status'] == 'PASS', 'Reused scalar-extension receipt is not PASS')
    return inputs, pins


def packet_binding_gate(inputs):
    ref=inputs['packet_binding_checker']
    path=(ROOT.parent/ref['path']).resolve()
    require(path.is_relative_to(ROOT.parent.resolve()),'Binding-checker path escapes repository')
    require(raw_digest(path)==ref['sha256'],'Packet binding checker changed')
    spec=importlib.util.spec_from_file_location('hodge_physical_source_packet_binding',path)
    require(spec is not None and spec.loader is not None,'Cannot load packet binding checker')
    binding=importlib.util.module_from_spec(spec)
    spec.loader.exec_module(binding)
    pinned={str(p.relative_to(ROOT.parent)):h for p,h in binding.PINS.items()}
    require(pinned==inputs['packet_binding_inputs'],'Binding inputs differ from fixed manifest')
    for name,expected in pinned.items():
        require(raw_digest(ROOT.parent/name)==expected,'Packet binding input changed: '+name)
    prior=json.loads(binding.EXACT_RECEIPT.read_text(encoding='utf-8'))
    require(prior['status']=='PASS' and len(prior['checks'])==10 and len(prior['mutation_controls'])==5,
        'Reused packet-bound exact-check scope differs')
    packet=json.loads(binding.PACKET.read_text(encoding='utf-8'))
    require(packet['convention']=='columns index input basis; coefficient derivative is partial_t+M_t',
        'Packet connection convention changed')
    text=(ROOT/(MODULE.replace('.','/')+'.lean')).read_text(encoding='utf-8')
    bindings=binding.bind(text)
    require(len(bindings)==11,'Incomplete literal packet/Lean definition binding')
    mutations=[]
    for label,original,replacement in [
        ('packet-row-sign','27*a*c*d0-12*b^2*d0+b*c^2','27*a*c*d0+12*b^2*d0+b*c^2'),
        ('packet-row-scalar','n0 a b c d0/(6*delta a b c d0)','n0 a b c d0/(12*delta a b c d0)'),
        ('packet-input-column','n0 a b c d0/(6*delta a b c d0)','n1 a b c d0/(6*delta a b c d0)')]:
        require(text.count(original)==1,'Ambiguous packet mutation target: '+label)
        try:
            binding.bind(text.replace(original,replacement))
        except RuntimeError as error:
            require(str(error).startswith('Lean expression mismatch:'),'Unrelated binding-control failure')
            mutations.append(label)
        else:
            raise RuntimeError('Packet mutation accepted: '+label)
    result=dict(status='PASS',executor=EXECUTOR,checker=ref,pinned_inputs=pinned,
        literal_definition_bindings=bindings,mutation_controls=mutations,
        reused_exact_receipt=str(binding.EXACT_RECEIPT.relative_to(ROOT.parent)),
        reused_exact_identities=10,reused_exact_mutations=5,reused_exact_check_rerun=False)
    return result
